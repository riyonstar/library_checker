# 【GitHubの画面で .github/workflows/deploy-worker.yml として作成して貼り付ける】
# 目的: バックエンド(Cloudflare Worker)を公開(デプロイ)します。手動実行(Run workflow)専用です。
# GitHub Secrets に必要なのは次の2つだけです(カーリルのAPIキーはここに置きません。Cloudflareの画面で登録します)。
#   CLOUDFLARE_API_TOKEN / CLOUDFLARE_ACCOUNT_ID
name: deploy-worker
on:
  workflow_dispatch: {}

jobs:
  deploy:
    runs-on: ubuntu-latest
    permissions:
      contents: read
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - name: 秘密情報の直書きチェック
        run: node check-secrets.mjs
      - name: 単体テスト
        run: node test-worker.mjs
      - name: Worker をデプロイ
        uses: cloudflare/wrangler-action@v3
        with:
          apiToken: ${{ secrets.CLOUDFLARE_API_TOKEN }}
          accountId: ${{ secrets.CLOUDFLARE_ACCOUNT_ID }}
